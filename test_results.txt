    function assert(name, condition, details = '') {
      const passed = Boolean(condition);
      results.push({ name, passed, details });
      const card = document.createElement('div');
      card.className = 'test-card ' + (passed ? 'pass' : 'fail');
      card.innerHTML = `
        <div style="display:flex; justify-content:space-between; align-items:center;">
          <strong>${passed ? 'Ô£à' : 'ÔØî'} ${name}</strong>
          <span class="badge ${passed ? 'pass' : 'fail'}">${passed ? 'PASS' : 'FAIL'}</span>
        </div>
        ${details ? `<div style="font-size: 0.85rem; color: #94a3b8; margin-top: 4px;">${details}</div>` : ''}
      `;
      testContainer.appendChild(card);
    }

      assert('DB: Omfattande ordf├Ârr├Ñd (>= 150 poster)', allItems && allItems.length >= 150, `Hittade ${allItems.length} poster`);
